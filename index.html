const USER_SHEET = 'Users';
const SCORE_SHEET = 'Scores';
const DASHBOARD_SHEET = 'Dashboard';

// 1. ฟังก์ชันสร้างตาราง Users, Scores และหน้า Dashboard อัตโนมัติจาก Google Sheet หน้าเปล่า
function setupChemLabDashboard() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  ensureSheetsExist(ss);
}

// ฟังก์ชันตรวจสอบและสร้างแผ่นงานทั้ง 3 แผ่นอัตโนมัติ
function ensureSheetsExist(ss) {
  // 1.1 แผ่นงาน Users สำหรับระบบลงทะเบียนและล็อกอินด้วยรหัสนักเรียน + รหัสผ่าน
  let userSheet = ss.getSheetByName(USER_SHEET);
  if (!userSheet) {
    const firstSheet = ss.getSheets()[0];
    if (firstSheet && firstSheet.getLastRow() === 0 && firstSheet.getName() !== SCORE_SHEET && firstSheet.getName() !== DASHBOARD_SHEET) {
      firstSheet.setName(USER_SHEET);
      userSheet = firstSheet;
    } else {
      userSheet = ss.insertSheet(USER_SHEET);
    }
  }
  if (userSheet.getLastRow() === 0) {
    userSheet.appendRow([
      'RegisteredAt',
      'StudentID',
      'Password',
      'FullName',
      'StudentNo',
      'ClassRoom',
      'Role',
      'LastLogin',
      'BestScore'
    ]);
    userSheet.getRange('A1:I1')
      .setBackground('#0F172A')
      .setFontColor('#00F5D4')
      .setFontWeight('bold');
    userSheet.setFrozenRows(1);
  }

  // 1.2 แผ่นงาน Scores สำหรับเก็บประวัติคะแนนและตัวชี้วัด (คอลัมน์ A ถึง L)
  let scoreSheet = ss.getSheetByName(SCORE_SHEET);
  if (!scoreSheet) {
    scoreSheet = ss.insertSheet(SCORE_SHEET);
  }
  if (scoreSheet.getLastRow() === 0) {
    scoreSheet.appendRow([
      'Timestamp',
      'StudentID',
      'PlayerName',
      'ClassRoom',
      'Role',
      'GameScore',
      'QuizScore',
      'TotalScore',
      'Std_M3_1_2',
      'Std_M3_3_4',
      'Std_M3_5',
      'Std_M3_6_7'
    ]);
    scoreSheet.getRange('A1:L1')
      .setBackground('#0B132B')
      .setFontColor('#FFFFFF')
      .setFontWeight('bold');
    scoreSheet.setFrozenRows(1);
  }

  // 1.3 แผ่นงาน Dashboard สรุปสถิติรายตัวชี้วัดและรายห้องเรียน (ม.3/1 ถึง ม.3/18)
  let dash = ss.getSheetByName(DASHBOARD_SHEET);
  if (!dash) {
    dash = ss.insertSheet(DASHBOARD_SHEET);
    buildDashboardLayout(dash);
  } else if (dash.getLastRow() === 0) {
    buildDashboardLayout(dash);
  }
  return { userSheet, scoreSheet, dash };
}

function buildDashboardLayout(dash) {
  dash.clear();
  dash.getRange('A1:H1').merge()
    .setValue('📊 CHEM-LAB RESCUE : DASHBOARD สรุปผลสัมฤทธิ์ทางการเรียนวิทยาศาสตร์ ม.3 (สสวท.)')
    .setBackground('#0B132B')
    .setFontColor('#00F5D4')
    .setFontSize(14)
    .setFontWeight('bold');

  // การ์ดสรุปภาพรวมด้านบน
  dash.getRange('A3:B3').setValues([['นักเรียนที่ลงทะเบียนทั้งหมด (คน)', '=COUNTA(Users!B2:B)']]);
  dash.getRange('C3:D3').setValues([['คะแนนสอบเฉลี่ยรวม (เต็ม 15)', '=IFERROR(ROUND(AVERAGE(Scores!G2:G), 2), 0)']]);
  dash.getRange('E3:F3').setValues([['คะแนนรวมสูงสุด (Top Score)', '=IFERROR(MAX(Scores!H2:H), 0)']]);
  dash.getRange('A3:F3').setBackground('#1C2541').setFontColor('#FFFFFF').setFontWeight('bold');

  // ตารางที่ 1: สรุปความก้าวหน้ารายตัวชี้วัด สสวท.
  dash.getRange('A5').setValue('1. สรุปผลการพิชิตภารกิจแยกตามกลุ่มตัวชี้วัด สสวท. (ว 2.1 ม.3/1 - ม.3/7)').setFontWeight('bold').setFontSize(12);
  dash.getRange('A6:D6').setValues([['รหัสตัวชี้วัด', 'สาระการเรียนรู้ (สสวท. ม.3 เล่ม 2)', 'จำนวนภารกิจที่พิชิตรวม', 'ค่าเฉลี่ยภารกิจต่อครั้ง']])
    .setBackground('#0F172A').setFontColor('#38BDF8').setFontWeight('bold');

  dash.getRange('A7:D10').setValues([
    ['ว 2.1 ม.3/1-2', 'วัสดุในชีวิตประจำวัน (พอลิเมอร์ เซรามิก วัสดุผสม และ 3Rs)', '=SUM(Scores!I2:I)', '=IFERROR(ROUND(AVERAGE(Scores!I2:I), 2), 0)'],
    ['ว 2.1 ม.3/3-4', 'กฎทรงมวลและปฏิกิริยาเคมี', '=SUM(Scores!J2:J)', '=IFERROR(ROUND(AVERAGE(Scores!J2:J), 2), 0)'],
    ['ว 2.1 ม.3/5', 'พลังงานความร้อนของปฏิกิริยา (ดูดความร้อน / คายความร้อน)', '=SUM(Scores!K2:K)', '=IFERROR(ROUND(AVERAGE(Scores!K2:K), 2), 0)'],
    ['ว 2.1 ม.3/6-7', 'ปฏิกิริยาเคมีในชีวิตประจำวัน (สนิม กรด-เบส การเผาไหม้)', '=SUM(Scores!L2:L)', '=IFERROR(ROUND(AVERAGE(Scores!L2:L), 2), 0)']
  ]);

  // ตารางที่ 2: สรุปผลสัมฤทธิ์แยกตามห้องเรียนครบทั้ง 18 ห้อง (ม.3/1 ถึง ม.3/18)
  dash.getRange('A12').setValue('2. สรุปสถิติแยกตามห้องเรียน (ม.3/1 ถึง ม.3/18)').setFontWeight('bold').setFontSize(12);
  dash.getRange('A13:E13').setValues([['ห้องเรียน', 'จำนวนครั้งที่ส่งผล', 'คะแนนเกมเฉลี่ย', 'คะแนนสอบเฉลี่ย (เต็ม 15)', 'คะแนนรวมสูงสุดของห้อง']])
    .setBackground('#0F172A').setFontColor('#FBBF24').setFontWeight('bold');

  const classRows = [];
  for (let i = 1; i <= 18; i++) {
    const cls = 'ม.3/' + i;
    classRows.push([
      cls,
      `=COUNTIF(Scores!D2:D, "${cls}")`,
      `=IFERROR(ROUND(AVERAGEIF(Scores!D2:D, "${cls}", Scores!F2:F), 0), 0)`,
      `=IFERROR(ROUND(AVERAGEIF(Scores!D2:D, "${cls}", Scores!G2:G), 2), 0)`,
      `=IFERROR(MAXIFS(Scores!H2:H, Scores!D2:D, "${cls}"), 0)`
    ]);
  }
  dash.getRange(14, 1, 18, 5).setValues(classRows);

  // ตารางที่ 3: Top 20 Leaderboard อัตโนมัติ
  dash.getRange('G5').setValue('🏆 อันดับคะแนนสูงสุด Top 20 (อัปเดตอัตโนมัติ)').setFontWeight('bold').setFontSize(12);
  dash.getRange('G6').setFormula('=IFERROR(QUERY(Scores!B1:H, "SELECT B, C, D, G, H WHERE B IS NOT NULL ORDER BY H DESC LIMIT 20", 1), "ยังไม่มีข้อมูล")');

  dash.autoResizeColumns(1, 12);
}

// 2. รับคำสั่งลงทะเบียน, ล็อกอิน และบันทึกคะแนนจากเว็บเกม (POST)
function doPost(e) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const { userSheet, scoreSheet } = ensureSheetsExist(ss);
    const d = JSON.parse(e.postData.contents);
    const action = d.action || 'submit_score';

    // กรณีลงทะเบียนใหม่ (Register)
    if (action === 'register') {
      const sid = String(d.studentId || '').trim();
      const rows = userSheet.getDataRange().getValues();
      for (let i = 1; i < rows.length; i++) {
        if (String(rows[i][1]).trim() === sid) {
          return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: 'รหัสนักเรียนนี้ถูกลงทะเบียนไว้แล้ว กรุณาเข้าสู่ระบบ' }))
            .setMimeType(ContentService.MimeType.JSON);
        }
      }
      userSheet.appendRow([
        new Date(),
        sid,
        String(d.password || ''),
        d.fullName || '',
        d.studentNo || '',
        d.classRoom || 'ม.3/1',
        d.role || 'ANALYST',
        new Date(),
        0
      ]);
      return ContentService.createTextOutput(JSON.stringify({
        status: 'success',
        user: { studentId: sid, fullName: d.fullName, studentNo: d.studentNo, classRoom: d.classRoom, role: d.role, bestScore: 0 }
      })).setMimeType(ContentService.MimeType.JSON);
    }

    // กรณีเข้าสู่ระบบ (Login)
    if (action === 'login') {
      const sid = String(d.studentId || '').trim();
      const pwd = String(d.password || '').trim();
      const rows = userSheet.getDataRange().getValues();
      for (let i = 1; i < rows.length; i++) {
        if (String(rows[i][1]).trim() === sid) {
          if (String(rows[i][2]).trim() !== pwd) {
            return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: 'รหัสผ่านไม่ถูกต้อง กรุณาลองอีกครั้ง' }))
              .setMimeType(ContentService.MimeType.JSON);
          }
          userSheet.getRange(i + 1, 8).setValue(new Date());
          return ContentService.createTextOutput(JSON.stringify({
            status: 'success',
            user: {
              studentId: sid,
              fullName: rows[i][3],
              studentNo: rows[i][4],
              classRoom: rows[i][5],
              role: rows[i][6],
              bestScore: Number(rows[i][8] || 0)
            }
          })).setMimeType(ContentService.MimeType.JSON);
        }
      }
      return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: 'ไม่พบรหัสนักเรียนนี้ในระบบ กรุณากดลงทะเบียนใหม่ก่อนเข้าเล่น' }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    // กรณีส่งคะแนนหลังเล่นเกมหรือทำแบบทดสอบ (submit_score)
    const sid = String(d.studentId || '-').trim();
    const gameScore = Number(d.gameScore || d.score || 0);
    const quizNum = Number(d.quizNumeric || 0);
    const totalScore = gameScore + (quizNum * 100);
    const displayPlayerName = d.studentNo ? `${d.name} (${d.studentNo})` : (d.name || 'ไม่ระบุชื่อ');

    scoreSheet.appendRow([
      new Date(),
      sid,
      displayPlayerName,
      d.classRoom || 'ม.3/1',
      d.role || '-',
      gameScore,
      quizNum,
      totalScore,
      Number(d.std_m3_1_2 || 0),
      Number(d.std_m3_3_4 || 0),
      Number(d.std_m3_5 || 0),
      Number(d.std_m3_6_7 || 0)
    ]);

    // อัปเดต BestScore ในชีต Users
    const uRows = userSheet.getDataRange().getValues();
    for (let i = 1; i < uRows.length; i++) {
      if (String(uRows[i][1]).trim() === sid) {
        const prevBest = Number(uRows[i][8] || 0);
        if (totalScore > prevBest) {
          userSheet.getRange(i + 1, 9).setValue(totalScore);
        }
        break;
      }
    }

    return ContentService.createTextOutput(JSON.stringify({ status: 'success', totalScore }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

// 3. ส่งข้อมูลขึ้นกระดาน Live Leaderboard หน้าห้องเรียน (GET)
function doGet(e) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const { scoreSheet } = ensureSheetsExist(ss);
    if (scoreSheet.getLastRow() <= 1) {
      return ContentService.createTextOutput(JSON.stringify([])).setMimeType(ContentService.MimeType.JSON);
    }
    const rows = scoreSheet.getDataRange().getValues();
    rows.shift();
    const list = rows.map(r => ({
      studentId: String(r[1] || '-'),
      name: r[2],
      classRoom: r[3],
      classInfo: r[3] + ' (' + r[4] + ')',
      gameScore: Number(r[5] || 0),
      quizScore: r[6] !== '' ? r[6] + '/15' : '-',
      score: Number(r[7] || r[5] || 0),
      stdSummary: `วัสดุ:${r[8]||0} | มวล:${r[9]||0} | พลังงาน:${r[10]||0} | ชีวิตจริง:${r[11]||0}`
    })).sort((a, b) => b.score - a.score).slice(0, 50);

    return ContentService.createTextOutput(JSON.stringify(list))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify([])).setMimeType(ContentService.MimeType.JSON);
  }
}

คลังความรู้วิทยาศาสตร์ ม.3 สสวท. (ปฏิกิริยาเคมีและวัสดุในชีวิตประจำวัน)

สรุปสูตร สมการเคมี และสมบัติวัสดุสำหรับใช้เปิดค้นคว้าระหว่างเล่นเกม

1. การเกิดปฏิกิริยาเคมี & กฎทรงมวล (ว 2.1 ม.3/3 – ม.3/4)

  • การจัดเรียงอะตอมใหม่: อะตอมของสารตั้งต้นไม่สูญหายและไม่เกิดขึ้นใหม่ แต่จัดเรียงตัวใหม่ได้ผลิตภัณฑ์
  • กฎทรงมวล (Law of Conservation of Mass): ในระบบปิด มวลรวมของสารตั้งต้น เท่ากับ มวลรวมของผลิตภัณฑ์เสมอ
  • ข้อควรระวังในระบบเปิด: หากปฏิกิริยาเกิดแก๊ส (เช่น $CO_2, H_2$) ในภาชนะเปิด แก๊สจะฟุ้งกระจายหนีออกไป ทำให้มวลที่ชั่งได้บนตาชั่งน้อยลง จึงต้องใช้ จุกยางปิดระบบ

2. พลังงานความร้อนของปฏิกิริยา (ว 2.1 ม.3/5)

  • ปฏิกิริยาคายความร้อน (Exothermic): ระบบถ่ายโอนความร้อนออกสู่สิ่งแวดล้อม ทำให้อุณหภูมิสิ่งแวดล้อมสูงขึ้น จับภาชนะแล้วรู้สึกร้อน (เช่น $CaCl_2$ หรือ $NaOH$ ละลายน้ำ, การเผาไหม้, ถุงประคบร้อน)
  • ปฏิกิริยาดูดความร้อน (Endothermic): ระบบดูดความร้อนจากสิ่งแวดล้อม ทำให้อุณหภูมิสิ่งแวดล้อมลดลง จับภาชนะแล้วรู้สึกเย็น (เช่น $NH_4NO_3$ ละลายน้ำ, การสังเคราะห์ด้วยแสง, ถุงประคบเย็น)

3. ปฏิกิริยาเคมีในชีวิตประจำวัน (ว 2.1 ม.3/6 – ม.3/7)

  • การเกิดสนิมเหล็ก: เหล็ก ($Fe$) + น้ำ ($H_2O$) + ออกซิเจน ($O_2$) ป้องกันโดยทาสี เคลือบน้ำมัน หรือชุบสังกะสี (กัลวาไนซ์)
  • กรด + โลหะ: ได้เกลือของโลหะ + แก๊สไฮโดรเจน ($H_2$ ติดไฟให้เสียงป๊อก)
  • กรด + คาร์บอเนต ($CaCO_3$): ได้เกลือ + น้ำ + แก๊สคาร์บอนไดออกไซด์ ($CO_2$) เช่น ฝนกรดกัดกร่อนหินอ่อน
  • การเผาไหม้ไม่สมบูรณ์: ออกซิเจนไม่พอ เกิดเขม่าควันดำและแก๊สพิษคาร์บอนมอนอกไซด์ ($CO$)

4. วัสดุในชีวิตประจำวันและหลัก 3Rs (ว 2.1 ม.3/1 – ม.3/2)

  • พอลิเมอร์: เทอร์โมพลาสติก (PET, PE หลอมซ้ำรีไซเคิลได้) vs เทอร์โมเซตติง (เบกาไลต์, เมลามีน ทนร้อนสูง ไม่หลอมซ้ำ)
  • เซรามิก: แข็ง ทนความร้อนสูง เป็นฉนวนความร้อนและไฟฟ้า แต่เปราะแตกง่าย
  • วัสดุผสม (Composite): รวมสมบัติเด่นของวัสดุตั้งแต่ 2 ชนิดขึ้นไป เช่น คาร์บอนไฟเบอร์ (CFRP), ไฟเบอร์กลาส, คอนกรีตเสริมเหล็ก
